Your account is authenticated by Supabase.

`, "Log out", async () => { const { error: logoutError } = await payworldSupabase.auth.signOut(); if (logoutError) { toast("Logout failed. Please try again."); return; } toast("You have logged out."); await updatePayworldAccount(); } ); return; } const root = $("modalRoot"); root.innerHTML = `

Sign in to Payworld

Sign in or create your Payworld account.
