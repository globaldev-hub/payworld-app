async function openAccountModal() { const { data: { session } } = await payworldSupabase.auth.getSession(); if (session) { const email = session.user.email || "Signed-in user"; showModal( "Your Payworld account", `

Signed in as ${escapeHtml(email)}

Your email account is authenticated by Supabase.

`, "Log out", async () => { const { error } = await payworldSupabase.auth.signOut(); if (error) { toast("Logout failed. Please try again."); return; } toast("You have logged out."); updatePayworldAccount(); } ); return; } const root = $("modalRoot"); root.innerHTML = `

Sign in to Payworld

Use your email to sign in or create an account.

`; let mode = "login"; $("authCancel").onclick = () => { root.innerHTML = ""; }; $("authToggle").onclick = () => { mode = mode === "login" ? "signup" : "login"; $("authTitle").textContent = mode === "login" ? "Sign in to Payworld" : "Create your account"; $("authSubmit").textContent = mode === "login" ? "Sign in" : "Create account"; $("authToggle").textContent = mode === "login" ? "Need an account? Sign up" : "Already registered? Sign in"; $("authPassword").autocomplete = mode === "login" ? "current-password" : "new-password"; $("authMessage").textContent = ""; }; $("payworldAuthForm").onsubmit = async (event) => { event.preventDefault(); const email = $("authEmail").value.trim(); const password = $("authPassword").value; const submit = $("authSubmit"); const message = $("authMessage"); submit.disabled = true; message.textContent = "Please wait…"; try { if (mode === "signup") { const { data, error } = await payworldSupabase.auth.signUp({ email, password, options: { emailRedirectTo: location.origin + location.pathname } }); if (error) throw error; if (data.session) { root.innerHTML = ""; toast("Your Payworld account has been created."); updatePayworldAccount(); } else { message.textContent = "Registration submitted. Check your email to confirm your account."; submit.disabled = false; } } else { const { error } = await payworldSupabase.auth.signInWithPassword({ email, password }); if (error) throw error; root.innerHTML = ""; toast("Welcome back to Payworld!"); updatePayworldAccount(); } } catch (error) { message.textContent = error.message || "Authentication failed."; submit.disabled = false; } }; } async function updatePayworldAccount() { const { data: { session } } = await payworldSupabase.auth.getSession(); const email = session?.user?.email; $("accountBtn").textContent = email ? "My account" : "Sign in"; $("accountStatus").textContent = email ? "Signed in as " + email : "You are browsing as a guest."; $("accountAction").textContent = email ? "Manage account" : "Sign in or create account"; } $("accountBtn").addEventListener("click", openAccountModal); $("accountAction").addEventListener("click", openAccountModal); payworldSupabase.auth.onAuthStateChange(() => { updatePayworldAccount(); });
